class PreviewActivity : AppCompatActivity() {

    private lateinit var rvItems: RecyclerView
    private lateinit var tvAddress: TextView
    private lateinit var tvPhone: TextView
    private lateinit var tvDateTime: TextView
    private lateinit var tvTotal: TextView
    private lateinit var btnPrint: Button
    private lateinit var printerPref: PrinterPreference

    // Bluetooth printing variables
    private var bluetoothAdapter: BluetoothAdapter? = null
    private var bluetoothSocket: BluetoothSocket? = null
    private var outputStream: OutputStream? = null
    private val printerExecutor = Executors.newSingleThreadExecutor()
    private var isPrinting = false

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_preview)

        printerPref = PrinterPreference(this)
        bluetoothAdapter = BluetoothAdapter.getDefaultAdapter()
        setupToolbar()
        bindViews()
        setupPreviewData()
    }

    private fun setupToolbar() {
        val toolbar = findViewById<MaterialToolbar>(R.id.toolbar)
        setSupportActionBar(toolbar)
    }

    private fun bindViews() {
        rvItems = findViewById(R.id.rvItems)
        tvAddress = findViewById(R.id.tvAddress)
        tvPhone = findViewById(R.id.tvPhone)
        tvDateTime = findViewById(R.id.tvDateTime)
        tvTotal = findViewById(R.id.tvTotal)
        btnPrint = findViewById(R.id.btnPrint)

        btnPrint.setOnClickListener {
            val printer = printerPref.getPrinter()
            if (printer == null) {
                showNoPrinterDialog()
            } else {
                executePrintJob(printer)
            }
        }
    }

    private fun setupPreviewData() {
        // Mock header data
        tvAddress.text = "No.12, Main Road, Yangon"
        tvPhone.text = "Phone: 09-123456789"
        tvDateTime.text = getCurrentDate()

        // Mock key-value data
        val items = listOf(
            PreviewItem("Coffee x2", "6,000"),
            PreviewItem("tea x2", "1,000"),
            PreviewItem("cold x2", "1,000"),
            PreviewItem("Cake x1", "2,500"),
            PreviewItem("Service Charge", "0")
        )

        rvItems.layoutManager = LinearLayoutManager(this)
        rvItems.adapter = PreviewAdapter(items)

        // Mock total
        tvTotal.text = "10,500"
    }

    private fun getCurrentDate(): String {
        val formatter = SimpleDateFormat("dd/MM/yyyy", Locale.getDefault())
        return formatter.format(Date())
    }

    @SuppressLint("MissingPermission")
    private fun executePrintJob(printerData: com.sa.webview.model.PrinterData) {
        if (isPrinting) {
            Toast.makeText(this, "Printing in progress...", Toast.LENGTH_SHORT).show()
            return
        }

        // Check Bluetooth
        if (bluetoothAdapter == null) {
            showErrorDialog("Error", "Device doesn't support Bluetooth")
            return
        }

        if (!bluetoothAdapter!!.isEnabled) {
            showErrorDialog("Bluetooth Off", "Please enable Bluetooth and try again")
            return
        }

        isPrinting = true
        btnPrint.isEnabled = false
        //btnPrint.text = "Printing..."

        printerExecutor.execute {
            try {
                // 1. Connect to printer
                val device = bluetoothAdapter?.getRemoteDevice(printerData.address)
                val pagerSize = printerData.paperSize
                if (device == null) {
                    throw PrintException("Printer not found: ${printerData.name}")
                }

                // Standard SPP UUID for serial communication
                val uuid = UUID.fromString("00001101-0000-1000-8000-00805F9B34FB")
                bluetoothSocket = device.createRfcommSocketToServiceRecord(uuid)

                bluetoothSocket?.connect()
                outputStream = bluetoothSocket?.outputStream

                // 2. Prepare receipt data
                //I want to check paper size and prepare Receipt data align format according to paper size
                val receiptData = prepareReceiptData()

                // 3. Send print commands
                printReceipt(receiptData, printerData.paperSize)

                // 4. Show success message
                runOnUiThread {
                    Toast.makeText(this, "Print job completed successfully", Toast.LENGTH_SHORT).show()
                }

            } catch (e: SecurityException) {
                Log.e("PrintJob", "Permission error: ${e.message}")
                runOnUiThread {
                    showErrorDialog("Permission Error", "Bluetooth permission required")
                }

            } catch (e: IOException) {
                Log.e("PrintJob", "Connection error: ${e.message}")
                runOnUiThread {
                    showErrorDialog("Connection Failed",
                        "Cannot connect to printer: ${printerData.name}\n" +
                                "Please check if printer is turned on and paired")
                }
                // Clear saved printer if connection fails
                printerPref.clearPrinter()

            } catch (e: PrintException) {
                Log.e("PrintJob", "Print error: ${e.message}")
                runOnUiThread {
                    showErrorDialog("Print Error", e.message ?: "Printing failed")
                }

            } catch (e: Exception) {
                Log.e("PrintJob", "Unexpected error: ${e.message}")
                runOnUiThread {
                    showErrorDialog("Print Failed",
                        "An unexpected error occurred: ${e.localizedMessage}")
                }

            } finally {
                // Cleanup
                try {
                    outputStream?.close()
                    bluetoothSocket?.close()
                } catch (e: IOException) {
                    Log.e("PrintJob", "Error closing connection: ${e.message}")
                }

                isPrinting = false
                runOnUiThread {
                    btnPrint.isEnabled = true
                    btnPrint.text = "PRINT"
                }
            }
        }
    }

    private fun prepareReceiptData(): List<String> {
        val receiptLines = mutableListOf<String>()

        // Get data from RecyclerView
        val adapter = rvItems.adapter as PreviewAdapter
        val items = adapter.getItems()

        // Header
        receiptLines.add("=".repeat(32))
        receiptLines.add("RECEIPT")
        receiptLines.add("=".repeat(32))
        receiptLines.add("")
        receiptLines.add(tvAddress.text.toString())
        receiptLines.add(tvPhone.text.toString())
        receiptLines.add("Date: ${tvDateTime.text}")
        receiptLines.add("-".repeat(32))
        receiptLines.add("ITEM                 AMOUNT")
        receiptLines.add("-".repeat(32))

        // Items
        items.forEach { item ->
            val line = String.format("%-20s %10s", item.key, item.value)
            receiptLines.add(line)
        }

        // Total
        receiptLines.add("-".repeat(32))
        receiptLines.add(String.format("%-20s %10s", "TOTAL", tvTotal.text))
        receiptLines.add("")
        receiptLines.add("=".repeat(32))
        receiptLines.add("    Thank you for your visit!")
        receiptLines.add("=".repeat(32))
        receiptLines.add("") // Empty line for cut position

        return receiptLines
    }

    private fun printReceipt(receiptLines: List<String>, paperSize: String) {
        outputStream?.let { os ->
            // Initialize printer
            os.write(byteArrayOf(0x1B, 0x40)) // ESC @ - Initialize

            // Set character code table (UTF-8)
            os.write(byteArrayOf(0x1B, 0x74, 0x10)) // ESC t 16 - UTF-8

            // Set alignment center for header
            os.write(byteArrayOf(0x1B, 0x61, 0x01)) // ESC a 1 - Center align

            // Set double height and width for header
            os.write(byteArrayOf(0x1D, 0x21, 0x11)) // GS ! 17 - Double height & width

            // Print each line
            receiptLines.forEachIndexed { index, line ->
                when {
                    index == 1 -> { // RECEIPT title
                        os.write(byteArrayOf(0x1B, 0x61, 0x01)) // Center align
                        os.write(byteArrayOf(0x1D, 0x21, 0x11)) // Double size
                        os.write("$line\n".toByteArray(Charsets.UTF_8))
                        os.write(byteArrayOf(0x1D, 0x21, 0x00)) // Normal size
                    }
                    index == 2 || index == receiptLines.size - 4 || index == receiptLines.size - 2 -> { // Divider lines
                        os.write(byteArrayOf(0x1B, 0x61, 0x01)) // Center align
                        os.write("$line\n".toByteArray(Charsets.UTF_8))
                    }
                    index == 8 || index == receiptLines.size - 7 -> { // "ITEM AMOUNT" and "TOTAL" lines
                        os.write(byteArrayOf(0x1B, 0x61, 0x00)) // Left align
                        os.write(byteArrayOf(0x1B, 0x45, 0x01)) // Bold on
                        os.write("$line\n".toByteArray(Charsets.UTF_8))
                        os.write(byteArrayOf(0x1B, 0x45, 0x00)) // Bold off
                    }
                    index >= 9 && index < receiptLines.size - 7 -> { // Items
                        os.write(byteArrayOf(0x1B, 0x61, 0x00)) // Left align
                        os.write("$line\n".toByteArray(Charsets.UTF_8))
                    }
                    else -> { // Other lines
                        os.write(byteArrayOf(0x1B, 0x61, 0x00)) // Left align
                        os.write("$line\n".toByteArray(Charsets.UTF_8))
                    }
                }

                // Add small delay for printer buffer
                Thread.sleep(10)
            }

            // Add line feeds and cut paper
            os.write(byteArrayOf(0x0A, 0x0A, 0x0A)) // 3 line feeds
            os.write(byteArrayOf(0x1D, 0x56, 0x41, 0x10)) // GS V A 16 - Partial cut

            os.flush()
        } ?: throw PrintException("No output stream available")
    }

    private fun showNoPrinterDialog() {
        AlertDialog.Builder(this)
            .setTitle("No Printer Selected")
            .setMessage("Please select a printer from the menu")
            .setPositiveButton("Select Printer") { _, _ ->
                val intent = Intent(this, PrinterListActivity::class.java)
                startActivity(intent)
            }
            .setNegativeButton("Cancel", null)
            .show()
    }

    private fun showErrorDialog(title: String, message: String) {
        runOnUiThread {
            AlertDialog.Builder(this)
                .setTitle(title)
                .setMessage(message)
                .setPositiveButton("OK", null)
                .show()
        }
    }

    // Custom exception class
    class PrintException(message: String) : Exception(message)

    override fun onCreateOptionsMenu(menu: Menu?): Boolean {
        menuInflater.inflate(R.menu.menu_printer, menu)
        return true
    }

    override fun onOptionsItemSelected(item: MenuItem): Boolean {
        return when (item.itemId) {
            R.id.action_printer -> {
                val intent = Intent(this, PrinterListActivity::class.java)
                startActivity(intent)
                true
            }
            else -> super.onOptionsItemSelected(item)
        }
    }

    override fun onDestroy() {
        super.onDestroy()
        // Cleanup resources
        try {
            outputStream?.close()
            bluetoothSocket?.close()
        } catch (e: IOException) {
            Log.e("PreviewActivity", "Error closing connections: ${e.message}")
        }
        printerExecutor.shutdown()
    }
}
