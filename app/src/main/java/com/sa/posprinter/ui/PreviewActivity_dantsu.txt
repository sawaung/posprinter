    private fun getCurrentDate(): String {
        val formatter = SimpleDateFormat("dd/MM/yyyy hh:mm a", Locale.getDefault())
        return formatter.format(Date())
    }

    @SuppressLint("MissingPermission")
    private fun executePrintJob(printerData: com.sa.webview.model.PrinterData) {
        if (isPrinting) {
            Toast.makeText(this, "Printing in progress...", Toast.LENGTH_SHORT).show()
            return
        }

        isPrinting = true
        btnPrint.isEnabled = false
        btnPrint.text = "Printing..."

        printerExecutor.execute {
            try {
                // Find the Bluetooth device
                val device = BluetoothAdapter.getDefaultAdapter()?.bondedDevices?.find {
                    it.address == printerData.address
                }

                if (device == null) {
                    throw PrintException("Printer not found: ${printerData.name}")
                }

                // Create Bluetooth connection
                val connection = BluetoothConnection(device)
                connection.connect()

                // Create printer instance with proper settings based on paper size
                val printer = createPrinter(connection, printerData.paperSize)

                // Print the receipt
                printReceipt(printer, printerData.paperSize)

                // Show success message
                runOnUiThread {
                    Toast.makeText(this, "Receipt printed successfully", Toast.LENGTH_SHORT).show()
                }

            } catch (e: EscPosConnectionException) {
                Log.e("PrintJob", "Connection error: ${e.message}")
                runOnUiThread {
                    showErrorDialog("Connection Failed",
                        "Cannot connect to printer: ${printerData.name}\n" +
                                "Please check if printer is turned on and paired")
                }
                // Clear saved printer if connection fails
                printerPref.clearPrinter()

            } catch (e: Exception) {
                Log.e("PrintJob", "Print error: ${e.message}")
                runOnUiThread {
                    showErrorDialog("Print Failed",
                        "An error occurred: ${e.localizedMessage}")
                }

            } finally {
                isPrinting = false
                runOnUiThread {
                    btnPrint.isEnabled = true
                    btnPrint.text = "PRINT"
                }
            }
        }
    }

    private fun createPrinter(
        connection: BluetoothConnection,
        paperSize: String
    ): EscPosPrinter {
        // Set printer density and width based on paper size
        return when (paperSize.toLowerCase()) {
            "58mm", "58" -> EscPosPrinter(
                connection,
                203,      // Printer dots density (203 DPI for 58mm)
                48.0f,    // Paper width in mm
                32        // Max characters per line
            )
            "76mm", "76" -> EscPosPrinter(
                connection,
                203,      // Printer dots density (203 DPI)
                76.0f,    // Paper width in mm
                42        // Max characters per line
            )
            "80mm", "80" -> EscPosPrinter(
                connection,
                203,      // Printer dots density (203 DPI)
                80.0f,    // Paper width in mm
                48        // Max characters per line
            )
            else -> EscPosPrinter(
                connection,
                203,
                80.0f,
                48
            )
        }
    }

    private fun printReceipt(printer: EscPosPrinter, paperSize: String) {
        // Get data from RecyclerView
        val adapter = rvItems.adapter as PreviewAdapter
        val items = adapter.getItems()

        // Calculate line width and formatting based on paper size
        val (maxChars, itemWidth, amountWidth) = when (paperSize.toLowerCase()) {
            "58mm", "58" -> Triple(32, 22, 8)
            "76mm", "76" -> Triple(42, 30, 10)
            "80mm", "80" -> Triple(48, 34, 12)
            else -> Triple(48, 34, 12)
        }

        // Build the receipt text
        val receiptText = buildString {
            // Store header
            appendLine("[C]<font size='big'><b>YOUR STORE NAME</b></font>")
            appendLine()
            appendLine("[C]${"=".repeat(maxChars)}")
            appendLine()

            // Store information
            appendLine("[C]${tvAddress.text}")
            appendLine("[C]${tvPhone.text}")
            appendLine("[C]Date: ${tvDateTime.text}")
            appendLine()

            // Receipt title
            appendLine("[C]<font size='big'><b>RECEIPT</b></font>")
            appendLine()
            appendLine("[C]${"-".repeat(maxChars)}")
            appendLine()

            // Column headers
            appendLine("<b>${"ITEM".padEnd(itemWidth)} ${"AMOUNT".padStart(amountWidth)}</b>")
            appendLine("[C]${"-".repeat(maxChars)}")
            appendLine()

            // Items list
            items.forEach { item ->
                val itemName = if (item.key.length > itemWidth) {
                    item.key.substring(0, itemWidth - 3) + "..."
                } else {
                    item.key.padEnd(itemWidth)
                }

                val amount = item.value.padStart(amountWidth)
                appendLine("$itemName $amount")
            }

            appendLine()
            appendLine("[C]${"-".repeat(maxChars)}")
            appendLine()

            // Total
            appendLine("<b><font size='big'>${"TOTAL".padEnd(itemWidth)} ${tvTotal.text.toString().padStart(amountWidth)}</font></b>")
            appendLine()
            appendLine()

            // Footer
            appendLine("[C]${"=".repeat(maxChars)}")
            appendLine("[C]<b>Thank you for your visit!</b>")
            appendLine("[C]Please come again")
            appendLine("[C]${"=".repeat(maxChars)}")

            // Paper cut command
            appendLine()
            appendLine()
            append("[C]<cut />")
        }

        // Print the receipt
        printer.printFormattedTextAndCut(receiptText)
    }

    // Alternative: More detailed receipt with logo
    private fun printDetailedReceipt(printer: EscPosPrinter, paperSize: String) {
        // Get data from RecyclerView
        val adapter = rvItems.adapter as PreviewAdapter
        val items = adapter.getItems()

        val maxChars = when (paperSize.toLowerCase()) {
            "58mm", "58" -> 32
            "76mm", "76" -> 42
            "80mm", "80" -> 48
            else -> 48
        }

        val receiptText = buildString {
            // Store logo (you can create a simple text-based logo)
            appendLine("[C]<font size='big'><b>★ STORE LOGO ★</b></font>")
            appendLine()

            // Store header
            appendLine("[C]<font size='big'>YOUR STORE NAME</font>")
            appendLine("[C]${"=".repeat(maxChars)}")
            appendLine()

            // Store details
            appendLine("[C]<font size='small'>${tvAddress.text}</font>")
            appendLine("[C]<font size='small'>${tvPhone.text}</font>")
            appendLine()
            appendLine("[C]<b>Date:</b> ${tvDateTime.text}")
            appendLine("[C]<b>Receipt No:</b> ${String.format("%06d", Random().nextInt(999999))}")
            appendLine()
            appendLine("[C]<font size='big'><b>--- RECEIPT ---</b></font>")
            appendLine()

            // Table header
            appendLine("[L]<b>${"No.".padEnd(3)} ${"Item".padEnd(22)} ${"Amount".padStart(8)}</b>")
            appendLine("[C]${"-".repeat(maxChars)}")
            appendLine()

            // Items with numbers
            items.forEachIndexed { index, item ->
                val itemNo = "${index + 1}.".padEnd(3)
                val itemName = if (item.key.length > 22) {
                    item.key.substring(0, 22 - 3) + "..."
                } else {
                    item.key.padEnd(22)
                }

                val amount = item.value.padStart(8)
                appendLine("[L]$itemNo $itemName $amount")
            }

            appendLine()
            appendLine("[C]${"-".repeat(maxChars)}")

            // Calculations
            val subtotal = 10500.0
            val tax = subtotal * 0.05
            val grandTotal = subtotal + tax

            appendLine("[L]${"Subtotal:".padEnd(30)} ${String.format("%,.0f", subtotal).padStart(8)}")
            appendLine("[L]${"Tax (5%):".padEnd(30)} ${String.format("%,.0f", tax).padStart(8)}")
            appendLine("[C]${"=".repeat(maxChars)}")
            appendLine("[L]<b><font size='big'>${"TOTAL:".padEnd(30)} ${String.format("%,.0f", grandTotal).padStart(8)}</font></b>")
            appendLine()
            appendLine()

            // Payment method
            appendLine("[L]<b>Payment Method:</b> CASH")
            appendLine("[L]<b>Change Due:</b> 0")
            appendLine()

            // Thank you message
            appendLine("[C]<b>${"*".repeat(maxChars)}</b>")
            appendLine("[C]<font size='big'><b>THANK YOU!</b></font>")
            appendLine("[C]We appreciate your business")
            appendLine("[C]<b>${"*".repeat(maxChars)}</b>")
            appendLine()
            appendLine("[C]<font size='small'>Terms & Conditions apply</font>")
            appendLine("[C]<font size='small'>Valid for 30 days</font>")

            // Paper cut
            appendLine()
            appendLine()
            append("[C]<cut />")
        }

        printer.printFormattedTextAndCut(receiptText)
    }

    private fun showNoPrinterDialog() {
        AlertDialog.Builder(this)
            .setTitle("No Printer Selected")
            .setMessage("Please select a printer from the menu")
            .setPositiveButton("Select Printer") { _, _ ->
                val intent = Intent(this, PrinterListActivity::class.java)
                startActivity(intent)
            }
            .setNegativeButton("Cancel", null)
            .show()
    }

    private fun showErrorDialog(title: String, message: String) {
        runOnUiThread {
            AlertDialog.Builder(this)
                .setTitle(title)
                .setMessage(message)
                .setPositiveButton("OK", null)
                .show()
        }
    }

    // Helper function to create a simple logo bitmap
    private fun createLogoBitmap(): Bitmap {
        val width = 200
        val height = 100
        val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)
        val canvas = Canvas(bitmap)

        // Draw background
        canvas.drawColor(Color.WHITE)

        // Draw text
        val paint = Paint().apply {
            color = Color.BLACK
            textSize = 36f
            typeface = Typeface.DEFAULT_BOLD
            isAntiAlias = true
            textAlign = Paint.Align.CENTER
        }

        canvas.drawText("STORE", width / 2f, height / 2f - 10, paint)
        paint.textSize = 24f
        canvas.drawText("LOGO", width / 2f, height / 2f + 30, paint)

        return bitmap
    }

    class PrintException(message: String) : Exception(message)

        menuInflater.inflate(R.menu.menu_printer, menu)
        return true
    }

    override fun onOptionsItemSelected(item: MenuItem): Boolean {
        return when (item.itemId) {
            R.id.action_printer -> {
                val intent = Intent(this, PrinterListActivity::class.java)
                startActivity(intent)
                true
            }
            else -> super.onOptionsItemSelected(item)
        }
    }

    override fun onDestroy() {
        super.onDestroy()
        printerExecutor.shutdown()
    }
