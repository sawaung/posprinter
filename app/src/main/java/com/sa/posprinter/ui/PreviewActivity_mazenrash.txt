package com.sa.webview.ui

import android.annotation.SuppressLint
import android.content.Intent
import android.os.Bundle
import android.util.Log
import android.view.Menu
import android.view.MenuItem
import android.widget.Button
import android.widget.TextView
import android.widget.Toast
import androidx.appcompat.app.AlertDialog
import androidx.appcompat.app.AppCompatActivity
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import com.google.android.material.appbar.MaterialToolbar
import com.mazenrashed.printooth.Printooth
import com.mazenrashed.printooth.data.printable.Printable
import com.mazenrashed.printooth.data.printable.TextPrintable
import com.mazenrashed.printooth.data.printer.DefaultPrinter
import com.mazenrashed.printooth.ui.ScanningActivity
import com.sa.webview.R
import com.sa.webview.model.PreviewItem
import com.sa.webview.model.PrinterData
import com.sa.webview.util.PrinterPreference
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import java.text.SimpleDateFormat
import java.util.*

class PreviewActivity : AppCompatActivity() {

    // UI Components
    private lateinit var rvItems: RecyclerView
    private lateinit var tvAddress: TextView
    private lateinit var tvPhone: TextView
    private lateinit var tvDateTime: TextView
    private lateinit var tvTotal: TextView
    private lateinit var btnPrint: Button

    // Data and preferences
    private lateinit var printerPref: PrinterPreference
    private var isPrinting = false

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_preview)

        // Initialize Printooth library
        Printooth.init(this)

        printerPref = PrinterPreference(this)
        setupToolbar()
        bindViews()
        setupPreviewData()
    }

    private fun setupToolbar() {
        val toolbar = findViewById<MaterialToolbar>(R.id.toolbar)
        setSupportActionBar(toolbar)
        supportActionBar?.setDisplayHomeAsUpEnabled(true)
    }

    private fun bindViews() {
        rvItems = findViewById(R.id.rvItems)
        tvAddress = findViewById(R.id.tvAddress)
        tvPhone = findViewById(R.id.tvPhone)
        tvDateTime = findViewById(R.id.tvDateTime)
        tvTotal = findViewById(R.id.tvTotal)
        btnPrint = findViewById(R.id.btnPrint)

        btnPrint.setOnClickListener {
            checkPrinterAndPrint()
        }
    }

    private fun setupPreviewData() {
        // Set store information
        tvAddress.text = "No.12, Main Road, Yangon"
        tvPhone.text = "Phone: 09-123456789"
        tvDateTime.text = getCurrentDate()

        // Create sample order items
        val items = listOf(
            PreviewItem("Coffee x2", "6,000"),
            PreviewItem("Tea x2", "1,000"),
            PreviewItem("Cold Drink x2", "1,000"),
            PreviewItem("Cake x1", "2,500"),
            PreviewItem("Service Charge", "0")
        )

        rvItems.layoutManager = LinearLayoutManager(this)
        rvItems.adapter = PreviewAdapter(items)

        // Calculate and display total
        tvTotal.text = "10,500"
    }

    private fun getCurrentDate(): String {
        val formatter = SimpleDateFormat("dd/MM/yyyy hh:mm a", Locale.getDefault())
        return formatter.format(Date())
    }

    private fun checkPrinterAndPrint() {
        val printerData = printerPref.getPrinter()

        if (printerData == null) {
            // No printer selected, show dialog to select one
            showNoPrinterDialog()
        } else {
            // Check if printer is paired
            if (isPrinterPaired(printerData)) {
                executePrintJob(printerData)
            } else {
                // Printer not paired, ask to pair again
                showPrinterNotPairedDialog(printerData)
            }
        }
    }

    private fun isPrinterPaired(printerData: PrinterData): Boolean {
        return Printooth.getPairedPrinter()?.address == printerData.address
    }

    @SuppressLint("MissingPermission")
    private fun executePrintJob(printerData: PrinterData) {
        // Prevent multiple print clicks
        if (isPrinting) {
            Toast.makeText(this, "Printing in progress...", Toast.LENGTH_SHORT).show()
            return
        }

        isPrinting = true
        btnPrint.isEnabled = false
        //btnPrint.text = "Printing..."

        // Use coroutines for background printing
        CoroutineScope(Dispatchers.IO).launch {
            try {
                val success = printReceipt(printerData)

                withContext(Dispatchers.Main) {
                    if (success) {
                        Toast.makeText(
                            this@PreviewActivity,
                            "Receipt printed successfully",
                            Toast.LENGTH_SHORT
                        ).show()
                    }
                }
            } catch (e: Exception) {
                Log.e("PrintJob", "Print error: ${e.message}", e)

                withContext(Dispatchers.Main) {
                    showErrorDialog(
                        "Print Failed",
                        "An error occurred while printing:\n${e.localizedMessage}"
                    )
                }
            } finally {
                withContext(Dispatchers.Main) {
                    isPrinting = false
                    btnPrint.isEnabled = true
                    btnPrint.text = "PRINT"
                }
            }
        }
    }

    private suspend fun printReceipt(printerData: PrinterData): Boolean {
        return withContext(Dispatchers.IO) {
            try {
                // Get data from RecyclerView
                val adapter = rvItems.adapter as? PreviewAdapter
                val items = adapter?.getItems() ?: emptyList()

                // Prepare printables based on paper size
                val printables = prepareReceiptPrintables(items, printerData.paperSize)
                //covert printables to arraylist


                // Print the receipt
                Printooth.printer().print(ArrayList<Printable>(printables))
                true
            } catch (e: Exception) {
                Log.e("PrintReceipt", "Error: ${e.message}", e)
                false
            }
        }
    }

    private fun prepareReceiptPrintables(
        items: List<Pair<String, String>>,
        paperSize: String
    ): List<Printable> {
        val printables = mutableListOf<Printable>()

        // Get paper width based on paper size
        val paperWidth = when (paperSize.lowercase(Locale.getDefault())) {
            "58mm", "58" -> 32
            "76mm", "76" -> 42
            "80mm", "80" -> 48
            else -> 48
        }

        // 1. Store Header
        printables.add(
            TextPrintable.Builder()
                .setText("YOUR STORE NAME")
                .setCharacterCode(DefaultPrinter.CHARCODE_PC1252)
                .setAlignment(DefaultPrinter.ALIGNMENT_CENTER)
                .setEmphasizedMode(DefaultPrinter.EMPHASIZED_MODE_BOLD)
                .setFontSize(DefaultPrinter.FONT_SIZE_LARGE)
                .setNewLinesAfter(1)
                .build()
        )

        // 2. Divider line
        printables.add(createDivider("-", paperWidth))

        // 3. Store Information
        printables.add(createCenteredText(tvAddress.text.toString()))
        printables.add(createCenteredText(tvPhone.text.toString()))
        printables.add(createCenteredText("Date: ${tvDateTime.text}"))

        // 4. Empty line
        printables.add(createEmptyLine())

        // 5. Receipt Title
        printables.add(
            TextPrintable.Builder()
                .setText("RECEIPT")
                .setCharacterCode(DefaultPrinter.CHARCODE_PC1252)
                .setAlignment(DefaultPrinter.ALIGNMENT_CENTER)
                .setEmphasizedMode(DefaultPrinter.EMPHASIZED_MODE_BOLD)
                .setFontSize(DefaultPrinter.FONT_SIZE_LARGE)
                .setNewLinesAfter(1)
                .build()
        )

        // 6. Divider line
        printables.add(createDivider("-", paperWidth))

        // 7. Table Headers
        val (itemWidth, amountWidth) = calculateColumnWidths(paperWidth)
        val headerText = String.format(
            "%-${itemWidth}s %${amountWidth}s",
            "ITEM",
            "AMOUNT"
        )

        printables.add(
            TextPrintable.Builder()
                .setText(headerText)
                .setCharacterCode(DefaultPrinter.CHARCODE_PC1252)
                .setAlignment(DefaultPrinter.ALIGNMENT_LEFT)
                .setEmphasizedMode(DefaultPrinter.EMPHASIZED_MODE_BOLD)
                .setNewLinesAfter(1)
                .build()
        )

        // 8. Divider line
        printables.add(createDivider("-", paperWidth))

        // 9. Order Items
        items.forEach { (itemName, price) ->
            val displayName = if (itemName.length > itemWidth) {
                itemName.substring(0, itemWidth - 3) + "..."
            } else {
                itemName
            }

            val itemText = String.format(
                "%-${itemWidth}s %${amountWidth}s",
                displayName,
                price
            )

            printables.add(
                TextPrintable.Builder()
                    .setText(itemText)
                    .setCharacterCode(DefaultPrinter.CHARCODE_PC1252)
                    .setAlignment(DefaultPrinter.ALIGNMENT_LEFT)
                    .setNewLinesAfter(1)
                    .build()
            )
        }

        // 10. Divider line
        printables.add(createDivider("-", paperWidth))

        // 11. Empty line
        printables.add(createEmptyLine())

        // 12. Total Amount
        val totalText = String.format(
            "%-${itemWidth}s %${amountWidth}s",
            "TOTAL",
            tvTotal.text.toString()
        )

        printables.add(
            TextPrintable.Builder()
                .setText(totalText)
                .setCharacterCode(DefaultPrinter.CHARCODE_PC1252)
                .setAlignment(DefaultPrinter.ALIGNMENT_LEFT)
                .setEmphasizedMode(DefaultPrinter.EMPHASIZED_MODE_BOLD)
                .setFontSize(DefaultPrinter.FONT_SIZE_LARGE)
                .setNewLinesAfter(2)
                .build()
        )

        // 13. Thank You Message
        printables.add(createDivider("=", paperWidth))
        printables.add(createCenteredText("Thank you for your visit!"))
        printables.add(createCenteredText("Please come again"))
        printables.add(createDivider("=", paperWidth))

        // 14. Paper Cut (add multiple empty lines before cut)
        repeat(3) {
            printables.add(createEmptyLine())
        }

        return printables
    }

    private fun createCenteredText(text: String): Printable {
        return TextPrintable.Builder()
            .setText(text)
            .setCharacterCode(DefaultPrinter.CHARCODE_PC1252)
            .setAlignment(DefaultPrinter.ALIGNMENT_CENTER)
            .setNewLinesAfter(1)
            .build()
    }

    private fun createDivider(char: String, length: Int): Printable {
        val divider = char.repeat(length)
        return TextPrintable.Builder()
            .setText(divider)
            .setCharacterCode(DefaultPrinter.CHARCODE_PC1252)
            .setAlignment(DefaultPrinter.ALIGNMENT_CENTER)
            .setNewLinesAfter(1)
            .build()
    }

    private fun createEmptyLine(): Printable {
        return TextPrintable.Builder()
            .setText("")
            .setNewLinesAfter(1)
            .build()
    }

    private fun calculateColumnWidths(paperWidth: Int): Pair<Int, Int> {
        return when (paperWidth) {
            32 -> Pair(22, 8)   // 58mm
            42 -> Pair(30, 10)  // 76mm
            48 -> Pair(34, 12)  // 80mm
            else -> Pair(34, 12) // Default to 80mm
        }
    }

    private fun showNoPrinterDialog() {
        AlertDialog.Builder(this)
            .setTitle("No Printer Selected")
            .setMessage("You need to select a printer first.")
            .setPositiveButton("Select Printer") { _, _ ->
                openPrinterSelection()
            }
            .setNegativeButton("Cancel", null)
            .show()
    }

    private fun showPrinterNotPairedDialog(printerData: PrinterData) {
        AlertDialog.Builder(this)
            .setTitle("Printer Not Connected")
            .setMessage("${printerData.name} is not connected. Do you want to pair it again?")
            .setPositiveButton("Pair Again") { _, _ ->
                openPrinterSelection()
            }
            .setNegativeButton("Cancel") { _, _ ->
                // Clear the saved printer since it's not paired
                printerPref.clearPrinter()
            }
            .show()
    }

    private fun openPrinterSelection() {
        val intent = Intent(this, ScanningActivity::class.java)
        startActivityForResult(intent, REQUEST_CODE_PRINTER_SELECTION)
    }

    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)

        if (requestCode == REQUEST_CODE_PRINTER_SELECTION && resultCode == RESULT_OK) {
            // Printer was selected successfully
            val printer = Printooth.getPairedPrinter()
            printer?.let {
                // Save printer to preferences
                val printerData = PrinterData(
                    name = it.name ?: "Unknown Printer",
                    address = it.address,
                    paperSize = "80mm", // Default paper size
                    isTested = false
                )
                printerPref.savePrinter(it.name!!,it.address,"80mm")

                Toast.makeText(this, "Printer paired successfully", Toast.LENGTH_SHORT).show()
            }
        }
    }

    private fun showErrorDialog(title: String, message: String) {
        AlertDialog.Builder(this)
            .setTitle(title)
            .setMessage(message)
            .setPositiveButton("OK", null)
            .show()
    }

    override fun onCreateOptionsMenu(menu: Menu?): Boolean {
        menuInflater.inflate(R.menu.menu_printer, menu)
        return true
    }

    override fun onOptionsItemSelected(item: MenuItem): Boolean {
        return when (item.itemId) {
            R.id.action_printer -> {
                openPrinterSelection()
                true
            }
            android.R.id.home -> {
                onBackPressed()
                true
            }
            else -> super.onOptionsItemSelected(item)
        }
    }

    companion object {
        private const val REQUEST_CODE_PRINTER_SELECTION = 1001
    }
}
